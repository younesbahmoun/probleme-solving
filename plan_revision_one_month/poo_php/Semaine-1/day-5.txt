        class CompteBancaire {
            private float $solde;

            public function deposer(float $montant) {
                $this->solde += $montant;
            }

            public function retirer($montant) {
                if ($this->solde >= $montant) {
                    $this->solde -= $montant;
                    return  $montant;
                }
            }
        }
